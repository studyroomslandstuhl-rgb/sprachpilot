 let media='';
 if(item.image)media=image(item.image,item.prompt);
 if(kind==='image-choice'&&item.audio)media+=`<div class="audio-panel"><button class="btn secondary" type="button" data-action="word-audio">🔊 Anhören</button></div>`;
 if(kind==='audio-produce')media+=textAudio(item);
 if(kind==='audio-file-choice'||kind==='audio-file-produce')media+=fileAudio(item);
 const prompt=kind==='cards'?'':`<h2 class="question">${esc(item.prompt||item.meaning||'Was ist richtig?')}</h2>`;
 let body='';
 if(kind==='choice'||kind==='audio-file-choice')body=options(item.options||[],item);
 else if(kind==='image-choice')body=imageOptions(item.options||[]);
 else if(kind==='input')body=answerInput('Schreibe die vollständige Antwort.');
 else if(kind==='order')body=orderBody(item);
 else if(kind==='speak'||kind==='produce'||kind==='audio-produce'||kind==='audio-file-produce')body=productionBody(kind==='produce'?'Pluralform mit Artikel':'Deine Antwort');
 else if(kind==='free')body=freeBody(item);
 else body=answerInput('Schreibe die Antwort.');
 return`<div class="question-card">${media}${prompt}${body}<div id="feedback" class="feedback">${help(item,task)}</div><div id="tech"></div></div>`
}
function options(list,item){
 const values=S.shuffle(list);
 const abc=values.length===3||item.abc===true;
 return`<div class="option-grid ${abc?'abc-list':''}">${values.map((value,index)=>`<button class="option" type="button" data-answer="${esc(value)}">${abc?`<span class="abc-letter">${String.fromCharCode(65+index)}</span>`:''}<span>${esc(value)}</span></button>`).join('')}</div>`
}

function bind(task,item){
 root.onclick=event=>{
  const button=event.target.closest('button');if(!button)return;
  if(button.dataset.answer!==undefined)return submit(button.dataset.answer);
  if(button.dataset.token!==undefined)return addToken(button);
  const action=button.dataset.action;
  if(action==='check-input')return submit(document.getElementById('answerInput')?.value||'');
  if(action==='check-order')return submit(R.order.join(' '));
  if(action==='undo-order')return undoOrder();
  if(action==='reset-order')return resetOrder();
  if(action==='open-write')return openWrite();
  if(action==='mic')return startMic(item);
  if(action==='card-audio')return S.sayWord(item.word,false,()=>technical('Die Audiofunktion ist nicht verfügbar.'));
  if(action==='word-audio')return S.sayWord(item.audio||item.answer,false,()=>technical('Die Audiofunktion ist nicht verfügbar.'));
  if(action==='text-audio')return cycleTextAudio(item);
  if(action==='card-next')return skipRevealedCard();
  if(action==='save-free')return saveFree(item);
  if(action==='repeat-exam')return repeatExam(task)
 };
 const card=document.getElementById('flipCard');
 card?.addEventListener('click',event=>{if(event.target.closest('button'))return;revealCard(item)});
 card?.addEventListener('keydown',event=>{if(event.key==='Enter'||event.key===' '){event.preventDefault();revealCard(item)}});
 document.getElementById('answerInput')?.addEventListener('keydown',event=>{if(event.key==='Enter')submit(event.target.value)});
 document.getElementById('freeAnswer')?.addEventListener('input',event=>saveDraft(event.target.value))
}
function submit(value){
 const task=R.task,item=task.items[R.index];
 if(R.busy||!String(value||'').trim())return;
 const ok=correct(item,value,task);
 S.attempt(task.id,task.items.length,R.index,ok);
 if(ok)return markRight();
 return task.exam?examWrong(item):normalWrong()
}
function normalWrong(){S.wrong(R.task.id,R.task.items.length);renderTask()}
function markRight(){
 const task=R.task,state=S.load(task.id,task.items.length),repeat=state.hadWrong||state.tries>0;
 R.busy=true;S.right(task.id,task.items.length,{forceDone:!!task.exam});
 disableArea();
 const feedback=document.getElementById('feedback');
 if(feedback)feedback.innerHTML=`<div class="ok">Richtig.${repeat&&!task.exam?' Die Aufgabe kommt am Ende noch einmal.':''}</div>`;
